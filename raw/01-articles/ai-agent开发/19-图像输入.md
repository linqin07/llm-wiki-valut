## 多模态 AI 简介

### 什么是多模态 AI

多模态 AI 是指能够同时处理多种类型输入（文本、图像、音频等）的人工智能系统。

```
传统文本模型：
  文本输入 --> [LLM] --> 文本输出

多模态模型：
  文本输入 --+
             |
  图像输入 --+--> [多模态 LLM] --> 文本输出
             |
  音频输入 --+
```

### 为什么需要图像输入

| 应用场景 | 说明 |
|---------|------|
| 图片描述 | 自动为图片生成描述文本 |
| OCR 文字识别 | 从图片中提取文字 |
| 图表分析 | 解读图表、数据可视化 |
| 文档理解 | 理解包含图表的文档 |
| 内容审核 | 审核图片内容是否合规 |
| 医学影像 | 辅助分析医学图片 |
| 商品识别 | 识别商品图片中的信息 |

```python
from langchain_core.messages import HumanMessage

def encode_image(image_path: str) -> str:
    """将图片文件编码为 base64 字符串"""
    with open(image_path, "rb") as f:
        return base64.b64encode(f.read()).decode("utf-8")

# 读取本地图片
image_path = "path/to/your/image.png"
base64_image = encode_image(image_path)

# 构造消息
message = HumanMessage(

                "url": f"data:image/png;base64,{base64_image}"
            }
        }
    ]
)

response = model.invoke([message])
```

### Base64 URL 格式

```
data:image/{格式};base64,{编码数据}

示例：
data:image/png;base64,iVBORw0KGgo...
data:image/jpeg;base64,/9j/4AAQ...
data:image/gif;base64,R0lGODlh...
```

### 自动检测图片格式

```python
import base64
import mimetypes

def image_to_base64_url(image_path: str) -> str:
    """将图片转为 base64 URL，自动检测格式"""
    # 检测 MIME 类型
    mime_type, _ = mimetypes.guess_type(image_path)
    if not mime_type:
        mime_type = "image/png"  # 默认

    # 编码
    with open(image_path, "rb") as f:
        base64_data = base64.b64encode(f.read()).decode("utf-8")

    return f"data:{mime_type};base64,{base64_data}"

# 使用
url = image_to_base64_url("photo.jpg")
message = HumanMessage(
    content=[
        {"type": "text", "text": "描述这张图片"},
        {"type": "image_url", "image_url": {"url": url}}
    ]
)
```

## 图像分析工具

### 为 Agent 添加图像分析能力

```python
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage
from langchain.chat_models import init_chat_model

model = init_chat_model("openai:gpt-4o")

@tool
def analyze_image(image_url: str, question: str) -> str:
    """分析图像并回答关于图像的问题。

    Args:
        image_url: 图片的URL地址
        question: 关于图片的问题
    """
    message = HumanMessage(
        content=[
            {"type": "text", "text": question},
            {"type": "image_url", "image_url": {"url": image_url}}
        ]
    )
    response = model.invoke([message])
    return response.content

@tool
def describe_image(image_url: str) -> str:
    """详细描述一张图片的内容。

    Args:
        image_url: 图片的URL地址
    """
    message = HumanMessage(
        content=[
            {"type": "text", "text": "请详细描述这张图片的内容，包括主要对象、颜色、场景等。"},
            {"type": "image_url", "image_url": {"url": image_url}}
        ]
    )
    response = model.invoke([message])
    return response.content

@tool
def extract_text_from_image(image_url: str) -> str:
    """从图片中提取文字（OCR）。

    Args:
        image_url: 图片的URL地址
    """
    message = HumanMessage(
        content=[
            {"type": "text", "text": "请提取这张图片中的所有文字内容，保持原有格式。"},
            {"type": "image_url", "image_url": {"url": image_url}}
        ]
    )
    response = model.invoke([message])
    return response.content
```

### 示例2：图像分析 Agent

```python
"""
图像分析 Agent
支持多种图像分析功能
"""
from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langchain_core.messages import BaseMessage, HumanMessage, AIMessage, SystemMessage
from langchain_core.tools import tool
from langchain.chat_models import init_chat_model

model = init_chat_model("openai:gpt-4o")

# ---- 工具定义 ----
@tool
def analyze_image(image_url: str, task: str) -> str:
    """分析图片。task 可以是 'describe', 'ocr', 'chart'。"""
    prompts = {
        "describe": "请详细描述这张图片的内容。",
        "ocr": "请提取图片中的所有文字，保持格式。",
        "chart": "请分析这张图表，说明数据趋势和关键信息。",
    }
    prompt = prompts.get(task, task)

    message = HumanMessage(
        content=[
            {"type": "text", "text": prompt},
            {"type": "image_url", "image_url": {"url": image_url}}
        ]
    )
    response = model.invoke([message])
    return response.content

# ---- 状态定义 ----
class VisionState(TypedDict):
    messages: Annotated[list[BaseMessage], add_messages]
    image_url: str
    analysis_result: str

# ---- 节点 ----
def analyze_node(state: VisionState) -> dict:
    """分析节点"""
    # 从消息中提取用户的分析请求
    last_msg = state["messages"][-1].content

    message = HumanMessage(
        content=[
            {"type": "text", "text": last_msg},
            {"type": "image_url", "image_url": {"url": state["image_url"]}}
        ]
    )
    response = model.invoke([
        SystemMessage(content="你是图像分析专家，请详细分析用户提供的图片。"),
        message
    ])
    return {
        "analysis_result": response.content,
        "messages": [AIMessage(content=response.content)],
    }

def format_node(state: VisionState) -> dict:
    """格式化输出"""
    formatted = f"## 图像分析结果\n\n{state['analysis_result']}"
    return {"messages": [AIMessage(content=formatted)]}

# ---- 构建图 ----
graph = StateGraph(VisionState)
graph.add_node("analyze", analyze_node)
graph.add_node("format", format_node)

graph.add_edge(START, "analyze")
graph.add_edge("analyze", "format")
graph.add_edge("format", END)

app = graph.compile()

# ---- 运行 ----
result = app.invoke({
    "messages": [HumanMessage(content="这张图片里有什么？请详细描述。")],
    "image_url": "https://example.com/sample.jpg",
    "analysis_result": "",
})
print(result["messages"][-1].content)
```

### 示例3：批量图片分析

```python
"""
批量图片分析
使用 Send 并行分析多张图片
"""
from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, START, END
from langgraph.types import Send
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langchain.chat_models import init_chat_model

model = init_chat_model("openai:gpt-4o")

class BatchState(TypedDict):
    image_urls: list[str]
    results: Annotated[list[str], lambda x, y: x + y]

class SingleImageState(TypedDict):
    image_url: str

def analyze_single(state: SingleImageState) -> dict:
    """分析单张图片"""
    message = HumanMessage(
        content=[
            {"type": "text", "text": "请简要描述这张图片的内容（50字以内）。"},
            {"type": "image_url", "image_url": {"url": state["image_url"]}}
        ]
    )
    response = model.invoke([message])
    return {"results": [f"{state['image_url']}: {response.content}"]}

def dispatch(state: BatchState) -> list:
    """分派每张图片给独立节点处理"""
    return [
        Send("analyze", {"image_url": url})
        for url in state["image_urls"]
    ]

def aggregate(state: BatchState) -> dict:
    """汇总结果"""
    summary = "\n".join(state["results"])
    return {"results": [f"=== 批量分析结果 ===\n{summary}"]}

# 构建图
graph = StateGraph(BatchState)
graph.add_node("analyze", analyze_single)
graph.add_node("aggregate", aggregate)

graph.add_conditional_edges(START, dispatch)
graph.add_edge("analyze", "aggregate")
graph.add_edge("aggregate", END)

app = graph.compile()

# 运行
result = app.invoke({
    "image_urls": [
        "https://example.com/image1.jpg",
        "https://example.com/image2.jpg",
        "https://example.com/image3.jpg",
    ],
    "results": [],
})
```

## 常见错误

### 1. 使用 Groq 进行图像分析

```python
# 错误：Groq 不支持图像
model = init_chat_model("groq:llama-3.3-70b-versatile")
message = HumanMessage(content=[
    {"type": "text", "text": "描述图片"},
    {"type": "image_url", "image_url": {"url": "..."}}
])
model.invoke([message])  # 会报错！

# 正确：使用支持视觉的模型
model = init_chat_model("openai:gpt-4o")
```

### 2. Base64 编码格式错误

```python
# 错误：缺少 data:image/ 前缀
url = f"base64,{base64_data}"

# 正确：完整的 data URL
url = f"data:image/png;base64,{base64_data}"
```

### 3. 图片路径不存在

```python
# 错误：硬编码的路径可能不存在
with open("/nonexistent/path.jpg", "rb") as f:  # FileNotFoundError!
    data = f.read()

# 正确：先检查文件存在
import os
image_path = "path/to/image.jpg"
if os.path.exists(image_path):
    with open(image_path, "rb") as f:
        data = f.read()
else:
    print(f"文件不存在: {image_path}")
```

### 4. 图片太大导致超时

```python
# 错误：直接使用超大图片
# 可能导致超时或 API 错误

# 正确：压缩或调整图片大小
from PIL import Image
import io

def resize_image(image_path: str, max_size: int = 1024) -> bytes:
    """调整图片大小"""
    img = Image.open(image_path)

    # 等比缩放
    ratio = max_size / max(img.size)
    if ratio < 1:
        new_size = tuple(int(x * ratio) for x in img.size)
        img = img.resize(new_size, Image.Resampling.LANCZOS)

    # 转为 bytes
    buffer = io.BytesIO()
    img.save(buffer, format="PNG")
    return buffer.getvalue()
```

```python
        img = img.resize(new_size, Image.Resampling.LANCZOS)

    # 转为 base64
    buffer = io.BytesIO()
    img.save(buffer, format="PNG", optimize=True)
    base64_data = base64.b64encode(buffer.getvalue()).decode("utf-8")

    return f"data:image/png;base64,{base64_data}"
```

### 4. 错误处理

```python
def safe_analyze_image(image_url: str) -> str:
    """带错误处理的图像分析"""
    try:
        message = HumanMessage(
            content=[
                {"type": "text", "text": "描述这张图片"},
                {"type": "image_url", "image_url": {"url": image_url}}
            ]
        )
        response = model.invoke([message])
        return response.content
    except Exception as e:
        if "invalid_image" in str(e).lower():
            return "错误：图片格式不支持或已损坏"
        elif "too_large" in str(e).lower():
            return "错误：图片文件太大"
        else:
            return f"分析失败: {str(e)}"
```
