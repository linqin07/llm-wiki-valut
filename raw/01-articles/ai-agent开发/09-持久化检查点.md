## 1. InMemorySaver 的局限性

### 内存存储的问题

InMemorySaver 将所有数据存储在 Python 进程的内存中，存在以下问题：

```
┌─────────────────────────────────────────────────────────────┐
│                InMemorySaver 的问题                          │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  问题1: 进程退出即丢失                                       │
│  ┌─────────────────────────────────────────────────────┐   │
│  │ 程序运行中: 保存了100个用户的对话历史                  │   │
│  │ 程序重启后: 所有历史全部丢失！                        │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                             │
│  问题2: 内存限制                                            │
│  ┌─────────────────────────────────────────────────────┐   │
│  │ 1000个用户 × 平均50条消息 × 每条1KB = 50MB           │   │
│  │ 10000个用户 = 500MB 内存占用                         │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                             │
│  问题3: 无法共享                                            │
│  ┌─────────────────────────────────────────────────────┐   │
│  │ 服务器A的InMemorySaver ≠ 服务器B的InMemorySaver      │   │
│  │ 无法实现负载均衡和高可用                              │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 何时使用 InMemorySaver

| 场景 | 是否适合 | 原因 |
|------|---------|------|
| 开发调试 | 适合 | 简单快速，无需配置 |
| 单元测试 | 适合 | 隔离测试，无副作用 |
| 演示Demo | 适合 | 无需数据库依赖 |
| 生产环境 | 不适合 | 数据不持久 |
| 多服务器 | 不适合 | 无法共享状态 |

## 2. SqliteSaver 持久化

### 基本导入和使用

```python
from langgraph.checkpoint.sqlite import SqliteSaver

# 创建SQLite检查点存储
# 重要：直接使用文件名，不需要sqlite:///前缀
with SqliteSaver.from_conn_string("checkpoints.sqlite") as checkpointer:
    # 在此范围内使用checkpointer
    pass
```

### 完整使用示例

```python
"""
SqliteSaver 持久化示例
"""
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent
from langgraph.checkpoint.sqlite import SqliteSaver


def sqlite_persistent_agent():
    """使用SQLite持久化的Agent"""

    model = init_chat_model("groq:llama-3.3-70b-versatile")

    # 使用SQLite持久化
    # 重要：使用with语句确保资源正确释放
    with SqliteSaver.from_conn_string("my_checkpoints.sqlite") as checkpointer:
        agent = create_agent(
            model=model,
            tools=[],
            system_prompt="你是一个有帮助的助手，请记住用户的信息。",
            checkpointer=checkpointer
        )

        config = {"configurable": {"thread_id": "persistent-user-1"}}

        # 第一轮对话
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我叫张三，我是程序员"}]},
            config=config
        )
        print(f"AI: {response['messages'][-1].content}")

        # 第二轮对话
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我住在北京"}]},
            config=config
        )
        print(f"AI: {response['messages'][-1].content}")

    print("\n程序退出，数据已保存到SQLite文件")


if __name__ == "__main__":
    sqlite_persistent_agent()
```

### SqliteSaver 的连接字符串格式

```python
# 正确格式（直接文件名）
SqliteSaver.from_conn_string("checkpoints.sqlite")
SqliteSaver.from_conn_string("data/my_checkpoints.db")
SqliteSaver.from_conn_string("/absolute/path/to/checkpoints.sqlite")

# 错误格式（不需要前缀）
# SqliteSaver.from_conn_string("sqlite:///checkpoints.sqlite")  # 错误！
```

### 使用上下文管理器

```python
# 推荐：使用with语句
with SqliteSaver.from_conn_string("checkpoints.sqlite") as checkpointer:
    agent = create_agent(
        model=model,
        tools=[],
        system_prompt="助手",
        checkpointer=checkpointer
    )
    # 使用agent...
# 退出with块时自动关闭连接

# 也可以手动管理（不推荐）
checkpointer = SqliteSaver.from_conn_string("checkpoints.sqlite")
try:
    agent = create_agent(
        model=model,
        tools=[],
        system_prompt="助手",
        checkpointer=checkpointer
    )
    # 使用agent...
finally:
    checkpointer.close()
```

## 4. 多用户会话管理

### 不同 thread_id 的状态隔离

```python
"""
多用户会话管理示例
"""
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent
from langgraph.checkpoint.sqlite import SqliteSaver


def multi_user_management():
    """演示多用户会话管理"""

    model = init_chat_model("groq:llama-3.3-70b-versatile")

    with SqliteSaver.from_conn_string("multi_user.sqlite") as checkpointer:
        agent = create_agent(
            model=model,
            tools=[],
            system_prompt="你是一个助手，请记住每个用户的信息。",
            checkpointer=checkpointer
        )

        # 用户A的对话
        config_a = {"configurable": {"thread_id": "user-a"}}
        print("用户A的对话:")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我叫小明，我是医生"}]},
            config=config_a
        )
        print(f"  用户A: 我叫小明，我是医生")
        print(f"  AI: {response['messages'][-1].content}")

        # 用户B的对话
        config_b = {"configurable": {"thread_id": "user-b"}}
        print("\n用户B的对话:")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我叫小红，我是老师"}]},
            config=config_b
        )
        print(f"  用户B: 我叫小红，我是老师")
        print(f"  AI: {response['messages'][-1].content}")

        # 用户C的对话
        config_c = {"configurable": {"thread_id": "user-c"}}
        print("\n用户C的对话:")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我叫小李，我是工程师"}]},
            config=config_c
        )
        print(f"  用户C: 我叫小李，我是工程师")
        print(f"  AI: {response['messages'][-1].content}")

        # 验证状态隔离
        print("\n" + "=" * 50)
        print("验证状态隔离")
        print("=" * 50)

        # 用户A问自己的职业
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我是做什么工作的？"}]},
            config=config_a
        )
        print(f"\n用户A问: 我是做什么工作的？")
        print(f"AI答: {response['messages'][-1].content}")

        # 用户B问自己的职业
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "我是做什么工作的？"}]},
            config=config_b
        )
        print(f"\n用户B问: 我是做什么工作的？")
        print(f"AI答: {response['messages'][-1].content}")


if __name__ == "__main__":
    multi_user_management()
```

### 查看所有会话

```python
"""
查看所有保存的会话
"""
from langgraph.checkpoint.sqlite import SqliteSaver


def list_all_sessions():
    """列出所有保存的会话"""

    with SqliteSaver.from_conn_string("multi_user.sqlite") as checkpointer:
        # 获取所有thread_id
        # 注意：InMemorySaver没有list方法，SqliteSaver有
        threads = checkpointer.list()

        print("所有保存的会话:")
        print("-" * 40)

        for thread in threads:
            config = {"configurable": {"thread_id": thread}}
            checkpoint = checkpointer.get(config)

            if checkpoint:
                messages = checkpoint.get("messages", [])
                print(f"Thread ID: {thread}")
                print(f"  消息数量: {len(messages)}")
                if messages:
                    last_msg = messages[-1]
                    print(f"  最后消息: {last_msg.content[:50]}...")
                print()


if __name__ == "__main__":
    list_all_sessions()
```

```python
"""
工具与持久化检查点结合示例
"""
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent
from langchain_core.tools import tool
from langgraph.checkpoint.sqlite import SqliteSaver


@tool
def get_weather(city: str) -> str:
    """获取指定城市的天气信息"""
    # 模拟天气API
    weather_data = {
        "北京": "晴天，25°C，湿度40%",
        "上海": "多云，22°C，湿度65%",
        "广州": "阵雨，28°C，湿度80%",
    }
    return weather_data.get(city, f"未找到{city}的天气信息")


@tool
def search_restaurants(city: str, cuisine: str) -> str:
    """搜索指定城市的餐厅"""
    # 模拟餐厅搜索
    restaurants = {
        ("北京", "中餐"): "全聚德烤鸭店、东来顺涮羊肉",
        ("北京", "西餐"): "莫斯科餐厅、马克西姆餐厅",
        ("上海", "中餐"): "南翔馒头店、老正兴菜馆",
    }
    key = (city, cuisine)
    return restaurants.get(key, f"未找到{city}的{cuisine}餐厅")


def tools_with_persistence():
    """工具与持久化结合"""

    model = init_chat_model("groq:llama-3.3-70b-versatile")

    with SqliteSaver.from_conn_string("tools_demo.sqlite") as checkpointer:
        agent = create_agent(
            model=model,
            tools=[get_weather, search_restaurants],
            system_prompt="你是一个旅行助手，可以帮助查询天气和餐厅信息。",
            checkpointer=checkpointer
        )

        config = {"configurable": {"thread_id": "travel-planning"}}

        # 第一轮：查询天气
        print("第1轮：查询北京天气")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "今天北京天气怎么样？"}]},
            config=config
        )
        print(f"AI: {response['messages'][-1].content}\n")

        # 第二轮：基于天气推荐活动
        print("第2轮：基于天气推荐活动")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "天气这么好，有什么推荐的活动吗？"}]},
            config=config
        )
        print(f"AI: {response['messages'][-1].content}\n")

        # 第三轮：查询餐厅
        print("第3轮：查询北京中餐厅")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "附近有什么好吃的中餐厅？"}]},
            config=config
        )
        print(f"AI: {response['messages'][-1].content}\n")

        # 第四轮：引用之前的工具结果
        print("第4轮：引用之前的工具结果")
        response = agent.invoke(
            {"messages": [{"role": "user", "content": "刚才查的那家烤鸭店怎么样？"}]},
            config=config
        )
        print(f"AI: {response['messages'][-1].content}\n")


if __name__ == "__main__":
    tools_with_persistence()
```

## 6. 检查点的内部结构

### State Snapshots 概念

检查点保存的是 **状态快照（State Snapshot）**，包含：

```
┌─────────────────────────────────────────────────────────────┐
│                    检查点内部结构                             │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  Checkpoint {                                               │
│    ├── id: "checkpoint-uuid"                                │
│    ├── ts: "2024-01-15T10:30:00Z"  (时间戳)                 │
│    ├── channel_values: {                                    │
│    │     ├── "messages": [                                  │
│    │     │     ├── SystemMessage { content: "..." }         │
│    │     │     ├── HumanMessage { content: "..." }          │
│    │     │     ├── AIMessage {                              │
│    │     │     │     ├── content: "..."                     │
│    │     │     │     └── tool_calls: [...]                  │
│    │     │     │   }                                        │
│    │     │     ├── ToolMessage {                            │
│    │     │     │     ├── name: "get_weather"                │
│    │     │     │     └── content: "..."                     │
│    │     │     │   }                                        │
│    │     │     └── AIMessage { content: "..." }             │
│    │     │   ]                                              │
│    │     └── "agent_state": { ... }                         │
│    │   }                                                    │
│    ├── channel_versions: { ... }                            │
│    └── versions_seen: { ... }                               │
│  }                                                          │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 检查点版本管理

```python
"""
检查点版本管理示例
"""
from langgraph.checkpoint.sqlite import SqliteSaver


def checkpoint_versioning():
    """演示检查点版本管理"""

    with SqliteSaver.from_conn_string("versioning_demo.sqlite") as checkpointer:
        # 获取某个thread的所有检查点
        config = {"configurable": {"thread_id": "version-test"}}

        # 列出所有检查点
        checkpoints = list(checkpointer.list(config))

        print(f"Thread 'version-test' 的检查点数量: {len(checkpoints)}")

        for i, checkpoint in enumerate(checkpoints):
            print(f"\n检查点 {i + 1}:")
            print(f"  ID: {checkpoint.id}")
            print(f"  时间: {checkpoint.ts}")

            messages = checkpoint.channel_values.get("messages", [])
            print(f"  消息数: {len(messages)}")


if __name__ == "__main__":
    checkpoint_versioning()
```

## 7. 生产环境考虑

### SQLite 的局限性

```
┌─────────────────────────────────────────────────────────────┐
│                SQLite 的局限性                                │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  1. 并发限制                                                │
│     - 写操作是串行的                                        │
│     - 高并发场景下性能下降                                   │
│                                                             │
│  2. 分布式限制                                              │
│     - 文件存储，无法跨服务器共享                              │
│     - 需要网络文件系统（NFS）才能多机访问                     │
│                                                             │
│  3. 扩展性限制                                              │
│     - 单机存储，容量有限                                     │
│     - 无法水平扩展                                          │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

| 特性 | SQLite | PostgreSQL | Redis |
|------|--------|------------|-------|
| 并发性能 | 低 | 高 | 高 |

## 8. 完整代码示例

### 示例1：完整的持久化 Agent

```python
"""
完整的持久化Agent示例
"""
import os
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent
from langchain_core.tools import tool
from langgraph.checkpoint.sqlite import SqliteSaver


@tool
def remember_preference(key: str, value: str) -> str:
    """保存用户偏好设置"""
    return f"已记住偏好: {key} = {value}"


@tool
def get_preference(key: str) -> str:
    """获取用户偏好设置"""
    # 在实际应用中，这里会查询数据库
    return f"偏好 {key} 的值为: 示例值"


def full_persistent_agent():
    """完整的持久化Agent示例"""

    model = init_chat_model("groq:llama-3.3-70b-versatile")

    # 确保数据目录存在
    os.makedirs("data", exist_ok=True)

    with SqliteSaver.from_conn_string("data/agent_checkpoints.sqlite") as checkpointer:
        agent = create_agent(
            model=model,
            tools=[remember_preference, get_preference],
            system_prompt="""你是一个智能助手。
你可以帮助用户记住他们的偏好设置。
请使用工具来保存和查询偏好。""",
            checkpointer=checkpointer
        )

        config = {"configurable": {"thread_id": "persistent-demo"}}

        # 模拟多轮对话
        conversations = [
            "你好，我想设置一些偏好",
            "我喜欢Python编程语言",
            "我偏好暗色主题",
            "我的时区是亚洲/上海",
            "你还记得我喜欢什么编程语言吗？",
```

### 示例2：会话管理器

```python
"""
会话管理器：管理多个用户的持久化会话
"""
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent
from langgraph.checkpoint.sqlite import SqliteSaver
from datetime import datetime


class SessionManager:
    """会话管理器"""

    def __init__(self, db_path: str = "sessions.sqlite"):
        self.db_path = db_path
        self.checkpointer = None
        self.agent = None

    def initialize(self):
        """初始化会话管理器"""
        model = init_chat_model("groq:llama-3.3-70b-versatile")
        self.checkpointer = SqliteSaver.from_conn_string(self.db_path)

        self.agent = create_agent(
            model=model,
            tools=[],
            system_prompt="你是一个有帮助的助手。",
            checkpointer=self.checkpointer
        )

    def get_session(self, user_id: str) -> dict:
        """获取用户的会话配置"""
        return {"configurable": {"thread_id": f"user-{user_id}"}}

    def chat(self, user_id: str, message: str) -> str:
        """处理用户消息"""
        config = self.get_session(user_id)
        response = self.agent.invoke(
            {"messages": [{"role": "user", "content": message}]},
            config=config
        )
        return response["messages"][-1].content

    def get_history(self, user_id: str) -> list:
        """获取用户的对话历史"""
        config = self.get_session(user_id)
        checkpoint = self.checkpointer.get(config)

        if checkpoint:
            return checkpoint.get("messages", [])
        return []

    def close(self):
        """关闭会话管理器"""
        if self.checkpointer:
            self.checkpointer.close()


def session_manager_example():
    """会话管理器示例"""

    manager = SessionManager("demo_sessions.sqlite")
    manager.initialize()

    try:
        # 用户A的对话
        print("用户A的对话:")
        print(f"  AI: {manager.chat('user-a', '你好，我叫小明')}")

        # 用户B的对话
        print("\n用户B的对话:")
        print(f"  AI: {manager.chat('user-b', '你好，我叫小红')}")

        # 用户A继续对话
        print("\n用户A继续对话:")
        print(f"  AI: {manager.chat('user-a', '你还记得我叫什么吗？')}")

        # 查看用户A的历史
        history = manager.get_history('user-a')
        print(f"\n用户A的历史消息数: {len(history)}")

    finally:
        manager.close()


if __name__ == "__main__":
    session_manager_example()
```

### 示例3：数据迁移工具

```python
"""
数据迁移工具：从InMemorySaver迁移到SqliteSaver
"""
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.checkpoint.sqlite import SqliteSaver


def migrate_data(source: InMemorySaver, target_path: str):
    """将数据从InMemorySaver迁移到SqliteSaver"""

    with SqliteSaver.from_conn_string(target_path) as target:
        # 获取所有thread
        threads = source.list()

        migrated_count = 0
        for thread in threads:
            config = {"configurable": {"thread_id": thread}}
            checkpoint = source.get(config)

            if checkpoint:
                # 保存到目标
                target.put(config, checkpoint)
                migrated_count += 1
                print(f"已迁移: {thread}")

        print(f"\n迁移完成，共迁移 {migrated_count} 个会话")


def migration_example():
    """数据迁移示例"""

    # 创建源数据
    source = InMemorySaver()

    # 模拟一些数据（实际使用中这些数据来自运行中的Agent）
    from langchain_core.messages import HumanMessage, AIMessage

    # 会话1
    config1 = {"configurable": {"thread_id": "session-1"}}
    source.put(config1, {
        "messages": [
            HumanMessage(content="你好"),
            AIMessage(content="你好！有什么可以帮助你的？")
        ]
    })

    # 会话2
    config2 = {"configurable": {"thread_id": "session-2"}}
    source.put(config2, {
        "messages": [
            HumanMessage(content="我叫张三"),
            AIMessage(content="你好张三！")
        ]
    })

    # 执行迁移
    print("开始数据迁移...")
    migrate_data(source, "migrated_data.sqlite")
    print("迁移完成！")


if __name__ == "__main__":
    migration_example()
```

## 9. 常见错误

### 错误1：使用错误的连接字符串格式

```python
# 错误：添加了sqlite:///前缀
checkpointer = SqliteSaver.from_conn_string("sqlite:///checkpoints.sqlite")
# 可能导致连接失败

# 正确：直接使用文件名
checkpointer = SqliteSaver.from_conn_string("checkpoints.sqlite")
```

### 错误2：忘记关闭连接

```python
# 错误：不使用with语句，可能忘记关闭
checkpointer = SqliteSaver.from_conn_string("checkpoints.sqlite")
agent = create_agent(model=model, tools=[], system_prompt="助手", checkpointer=checkpointer)
# 使用agent...
# 如果忘记checkpointer.close()，可能导致数据未完全写入

# 正确：使用with语句
with SqliteSaver.from_conn_string("checkpoints.sqlite") as checkpointer:
    agent = create_agent(model=model, tools=[], system_prompt="助手", checkpointer=checkpointer)
    # 使用agent...
# 自动关闭
```

### 错误3：路径问题

```python
# 错误：相对路径可能因工作目录变化而出错
checkpointer = SqliteSaver.from_conn_string("checkpoints.sqlite")

# 正确：使用绝对路径或确保目录存在
import os
os.makedirs("data", exist_ok=True)
checkpointer = SqliteSaver.from_conn_string("data/checkpoints.sqlite")
```

### 错误4：并发访问问题

```python
# 错误：多个进程同时写入同一个SQLite文件
# 可能导致数据库锁定错误

# 正确：对于高并发场景，使用PostgreSQL或Redis
```

```python
def backup_database(db_path: str, backup_dir: str = "backups"):
    """备份数据库"""
    os.makedirs(backup_dir, exist_ok=True)
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
```

### 3. 清理过期数据

```python
def cleanup_old_sessions(db_path: str, days: int = 30):
    """清理超过指定天数的会话"""
    import sqlite3
    from datetime import datetime, timedelta

    cutoff = datetime.now() - timedelta(days=days)

    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()

    # 删除旧的检查点
    cursor.execute("""
        DELETE FROM checkpoints
        WHERE created_at < ?
    """, (cutoff.isoformat(),))

    deleted = cursor.rowcount
    conn.commit()
    conn.close()

    print(f"已清理 {deleted} 个过期检查点")
```

```python
def get_database_size(db_path: str) -> dict:
    """获取数据库大小信息"""
    import os
    import sqlite3

    file_size = os.path.getsize(db_path)
    file_size_mb = file_size / (1024 * 1024)

    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()

    cursor.execute("SELECT COUNT(*) FROM checkpoints")
    checkpoint_count = cursor.fetchone()[0]

    conn.close()

    return {
        "file_size_mb": round(file_size_mb, 2),
        "checkpoint_count": checkpoint_count,
    }
```
